cmake_minimum_required(VERSION 3.16)
project(AtlasKV VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(atlaskv src/database.cpp)
target_include_directories(atlaskv PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(atlaskv PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(atlaskv_cli src/main.cpp)
target_link_libraries(atlaskv_cli PRIVATE atlaskv)
add_executable(atlaskv_tests tests/test_database.cpp)
target_link_libraries(atlaskv_tests PRIVATE atlaskv Threads::Threads)
enable_testing()
add_test(NAME atlaskv_tests COMMAND atlaskv_tests)
add_executable(atlaskv_bench benchmarks/bench.cpp)
target_link_libraries(atlaskv_bench PRIVATE atlaskv)
